# Gepinnter Legacy-Stack NUR für den Windows-7-kompatiblen Release-Build.
#
# Windows 7 wird von PySide6/Qt6 grundsätzlich nicht unterstützt (nie
# unterstützt, keine Frage der Version) und von CPython selbst nur noch bis
# Python 3.8 (ab 3.9 offiziell "cannot be used on Windows 7 or earlier").
# Dieser Stack ist daher bewusst alt und unsupportet (Python 3.8 ist seit
# 10/2024 EOL, Qt 5.15 Open-Source-Branch erhält keine Fixes mehr). Nur für
# die Ziel-Laborrechner verwenden, die wirklich auf Windows 7 laufen -- für
# alles andere (Entwicklung, moderne Windows-10/11-Rechner) gilt weiterhin
# pyproject.toml mit PySide6.
#
# Versionen wurden gezielt gegen Python 3.8 aufgelöst und lokal getestet
# (Laden, ROI-Handling, Export, Docking, Wiedergabe) -- ein Testlauf auf
# echter Windows-7-Hardware vor dem Rollout wird trotzdem empfohlen.
PyQt5==5.15.11
numpy==1.24.4
pyqtgraph==0.13.3
pyinstaller==5.13.2
qtpy==2.4.3
pytest==7.4.4

# imageio/imageio-ffmpeg (Video-Export): die in pyproject.toml fuer den
# modernen Stack gepinnte Version (imageio>=2.37.4) setzt Python>=3.10 voraus
# und liesse sich unter Python 3.8 (Windows 7) gar nicht installieren -- hier
# stattdessen die letzte zu Python 3.8 kompatible Version gepinnt. Geprueft:
# echter imageio.get_writer(...)-Videoschreibvorgang (inkl. mitgeliefertem
# ffmpeg-Binary) funktioniert unter diesem Stack einwandfrei. main_window.py
# importiert das Paket weiterhin nur lazy innerhalb der Export-Funktion, d.h.
# FEHLT es doch einmal (z.B. bei einer manuell unvollstaendigen Installation),
# zeigt der Video-Export weiterhin einen Fehlerdialog statt abzustuerzen.
imageio==2.35.1
imageio-ffmpeg==0.5.1
# pillow (transitive Abhaengigkeit von imageio) explizit gepinnt: die sonst
# aufgeloeste neueste zu Python 3.8 kompatible Version (10.4.0) importiert
# beim PyInstaller-Bau (ueber dessen Splash-Screen-Unterstuetzung, die PIL
# lädt) numpy.typing.NDArray, was in der hier gepinnten numpy-Version
# (1.24.4) nicht existiert -- Bugfix: "AttributeError: module 'numpy.typing'
# has no attribute 'NDArray'" beim exe-Bau. Eine aeltere Pillow-Version ohne
# diese Typannotation umgeht das; fuer imageios eigentliche Verwendung
# (Bild-/Videoframes lesen/schreiben) reicht sie unveraendert aus.
pillow==9.5.0

# tifffile (TIFF-Import, siehe data.py load_tiff_grayscale/
# tiff_crop_to_temperature): die per pyproject.toml aufgeloeste aktuelle
# Version ist unter Python 3.8 nicht installierbar -- hier die letzte dazu
# kompatible Version gepinnt. Der zugehoerige Menüpunkt ist derzeit
# bewusst deaktiviert/ausgegraut (siehe ui_build.py), der Code UND seine
# Tests bleiben aber auf allen Plattformen funktionsfaehig.
tifffile==2023.7.10
